이벤트 함수

버튼의 click 이벤트에 함수를 연결합니다. 버튼을 누를 때마다 횟수가 증가합니다.

실행 결과

버튼을 눌러 실행하세요.

예제 코드

let count = 0;
function handleClick() {
    count += 1;
    document.getElementById("result").textContent = "클릭 횟수: " + count;
}
document.getElementById("run").addEventListener("click", handleClick);

함수 (Function)

함수는 호출할 때 실행됩니다. 두 숫자를 매개변수로 받아 더한 값을 반환합니다.

실행 결과

버튼을 눌러 실행하세요.

예제 코드

function add(a, b) {
    return a + b;
}
function runExample() {
    const a = Number(document.getElementById("a").value);
    const b = Number(document.getElementById("b").value);
    document.getElementById("result").textContent = "합계: " + add(a, b);
}
document.getElementById("run").addEventListener("click", runExample);

내장 함수 (prompt, confirm, alert)

prompt는 입력, confirm은 확인 또는 취소, alert는 알림 창을 표시합니다.

실행 결과

버튼을 눌러 실행하세요.

예제 코드

function runExample() {
    const name = prompt("이름을 입력하세요.");
    if (name === null) {
        document.getElementById("result").textContent = "입력을 취소했습니다.";
        return;
    }
    const accepted = confirm(name + "님, 알림을 표시할까요?");
    if (accepted) {
        alert(name + "님, 안녕하세요!");
    }
    document.getElementById("result").textContent = accepted ? "확인을 선택했습니다." : "취소를 선택했습니다.";
}
document.getElementById("run").addEventListener("click", runExample);

if-else 조건문

입력한 정수가 짝수인지 홀수인지 조건에 따라 구분합니다.

실행 결과

버튼을 눌러 실행하세요.

예제 코드

function runExample() {
    const input = document.getElementById("number").value;
    const number = Number(input);
    let message;
    if (input === "" || !Number.isSafeInteger(number)) {
        message = "안전한 범위의 정수를 입력하세요.";
    } else {
        if (number % 2 === 0) {
            message = "짝수입니다.";
        } else {
            message = "홀수입니다.";
        }
    }
    document.getElementById("result").textContent = message;
}
document.getElementById("run").addEventListener("click", runExample);

else if 조건문

0~100점의 점수를 입력하면 여러 조건을 차례로 확인하여 등급을 결정합니다.

실행 결과

버튼을 눌러 실행하세요.

예제 코드

function runExample() {
    const input = document.getElementById("score").value;
    const score = Number(input);
    let grade;
    if (input === "" || !Number.isFinite(score) || score < 0 || score > 100) {
        grade = "0~100점 사이로 입력하세요.";
    } else if (score >= 90) {
        grade = "A";
    } else if (score >= 80) {
        grade = "B";
    } else if (score >= 70) {
        grade = "C";
    } else {
        grade = "D";
    }
    document.getElementById("result").textContent = "결과: " + grade;
}
document.getElementById("run").addEventListener("click", runExample);

switch-case 조건문

선택값과 일치하는 case를 실행합니다. break로 switch문을 종료합니다.

실행 결과

버튼을 눌러 실행하세요.

예제 코드

function runExample() {
    const day = document.getElementById("day").value;
    let message;
    switch (day) {
        case "월":
            message = "한 주의 시작입니다.";
            break;
        case "금":
            message = "금요일입니다.";
            break;
        case "토":
        case "일":
            message = "주말입니다.";
            break;
        default:
            message = "평일입니다.";
    }
    document.getElementById("result").textContent = message;
}
document.getElementById("run").addEventListener("click", runExample);

for 반복문

초기값, 조건, 증감을 지정하여 1부터 5까지 출력하고 합계를 계산합니다.

실행 결과

버튼을 눌러 실행하세요.

예제 코드

function runExample() {
    const lines = [];
    let sum = 0;
    for (let i = 1; i <= 5; i++) {
        lines.push(String(i));
        sum += i;
    }
    document.getElementById("result").textContent = lines.join(" → ") + "\n합계: " + sum;
}
document.getElementById("run").addEventListener("click", runExample);

이중 for문

바깥 반복문은 단을, 안쪽 반복문은 곱하는 수를 바꾸어 2~4단을 출력합니다.

실행 결과

버튼을 눌러 실행하세요.

예제 코드

function runExample() {
    const lines = [];
    for (let dan = 2; dan <= 4; dan++) {
        for (let i = 1; i <= 9; i++) {
            lines.push(dan + " × " + i + " = " + dan * i);
        }
        lines.push("");
    }
    document.getElementById("result").textContent = lines.join("\n");
}
document.getElementById("run").addEventListener("click", runExample);

while 반복문

조건을 먼저 검사합니다. 시작값이 5보다 크면 반복하지 않습니다.

실행 결과

버튼을 눌러 실행하세요.

예제 코드

function runExample() {
    const input = document.getElementById("start").value;
    let i = Number(input);
    if (input === "" || !Number.isInteger(i) || i < 1 || i > 6) {
        document.getElementById("result").textContent = "1~6 사이의 정수를 입력하세요.";
        return;
    }
    const lines = [];
    while (i <= 5) {
        lines.push(String(i));
        i++;
    }
    document.getElementById("result").textContent = lines.length ? lines.join(" → ") : "조건이 거짓이므로 실행 횟수는 0회입니다.";
}
document.getElementById("run").addEventListener("click", runExample);

do-while 반복문

본문을 실행한 뒤 조건을 검사합니다. 시작값이 6이어도 본문을 한 번 실행합니다.

실행 결과

버튼을 눌러 실행하세요.

예제 코드

function runExample() {
    const input = document.getElementById("start").value;
    let i = Number(input);
    if (input === "" || !Number.isInteger(i) || i < 1 || i > 6) {
        document.getElementById("result").textContent = "1~6 사이의 정수를 입력하세요.";
        return;
    }
    const lines = [];
    do {
        lines.push(String(i));
        i++;
    } while (i <= 5);
    document.getElementById("result").textContent = lines.join(" → ") + "\n실행 횟수: " + lines.length + "회";
}
document.getElementById("run").addEventListener("click", runExample);
