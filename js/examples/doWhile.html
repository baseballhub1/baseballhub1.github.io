do-while 반복문

본문을 실행한 뒤 조건을 검사합니다. 시작값이 6이어도 본문을 한 번 실행합니다.

실행 결과

버튼을 눌러 실행하세요.

예제 코드

function runExample() {
    const input = document.getElementById("start").value;
    let i = Number(input);
    if (input === "" || !Number.isInteger(i) || i < 1 || i > 6) {
        document.getElementById("result").textContent = "1~6 사이의 정수를 입력하세요.";
        return;
    }
    const lines = [];
    do {
        lines.push(String(i));
        i++;
    } while (i <= 5);
    document.getElementById("result").textContent = lines.join(" → ") + "\n실행 횟수: " + lines.length + "회";
}
document.getElementById("run").addEventListener("click", runExample);
