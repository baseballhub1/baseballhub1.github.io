else if 조건문

0~100점의 점수를 입력하면 여러 조건을 차례로 확인하여 등급을 결정합니다.

실행 결과

버튼을 눌러 실행하세요.

예제 코드

function runExample() {
    const input = document.getElementById("score").value;
    const score = Number(input);
    let grade;
    if (input === "" || !Number.isFinite(score) || score < 0 || score > 100) {
        grade = "0~100점 사이로 입력하세요.";
    } else if (score >= 90) {
        grade = "A";
    } else if (score >= 80) {
        grade = "B";
    } else if (score >= 70) {
        grade = "C";
    } else {
        grade = "D";
    }
    document.getElementById("result").textContent = "결과: " + grade;
}
document.getElementById("run").addEventListener("click", runExample);
