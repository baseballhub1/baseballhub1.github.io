if-else 조건문

입력한 정수가 짝수인지 홀수인지 조건에 따라 구분합니다.

실행 결과

버튼을 눌러 실행하세요.

예제 코드

function runExample() {
    const input = document.getElementById("number").value;
    const number = Number(input);
    let message;
    if (input === "" || !Number.isSafeInteger(number)) {
        message = "안전한 범위의 정수를 입력하세요.";
    } else {
        if (number % 2 === 0) {
            message = "짝수입니다.";
        } else {
            message = "홀수입니다.";
        }
    }
    document.getElementById("result").textContent = message;
}
document.getElementById("run").addEventListener("click", runExample);
