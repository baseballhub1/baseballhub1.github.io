내장 함수 (prompt, confirm, alert)

prompt는 입력, confirm은 확인 또는 취소, alert는 알림 창을 표시합니다.

실행 결과

버튼을 눌러 실행하세요.

예제 코드

function runExample() {
    const name = prompt("이름을 입력하세요.");
    if (name === null) {
        document.getElementById("result").textContent = "입력을 취소했습니다.";
        return;
    }
    const accepted = confirm(name + "님, 알림을 표시할까요?");
    if (accepted) {
        alert(name + "님, 안녕하세요!");
    }
    document.getElementById("result").textContent = accepted ? "확인을 선택했습니다." : "취소를 선택했습니다.";
}
document.getElementById("run").addEventListener("click", runExample);
