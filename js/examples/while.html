while 반복문

조건을 먼저 검사합니다. 시작값이 5보다 크면 반복하지 않습니다.

실행 결과

버튼을 눌러 실행하세요.

예제 코드

function runExample() {
    const input = document.getElementById("start").value;
    let i = Number(input);
    if (input === "" || !Number.isInteger(i) || i < 1 || i > 6) {
        document.getElementById("result").textContent = "1~6 사이의 정수를 입력하세요.";
        return;
    }
    const lines = [];
    while (i <= 5) {
        lines.push(String(i));
        i++;
    }
    document.getElementById("result").textContent = lines.length ? lines.join(" → ") : "조건이 거짓이므로 실행 횟수는 0회입니다.";
}
document.getElementById("run").addEventListener("click", runExample);
